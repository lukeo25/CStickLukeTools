Wick/Candlestick Recovery

Universal autosave recovery for Wick Editor and Candlestick

⚠️ Important: Run this on the correct site!

IndexedDB is domain-specific. This tool must be run from the same domain as your editor:

• Wick Editor: Open this file, then go to wickeditor.com and paste the scan code in console
• Candlestick: Open this file, then go to candlestick.gg and paste the scan code in console

Or use the Bookmarklet below directly on the editor site!

📋 Option 1: Use Console Script (Recommended)

1. Go to wickeditor.com or candlestick.gg in your browser
2. Open Developer Tools (F12 or Ctrl+Shift+I)
3. Go to the Console tab
4. Paste and run this code: 

// Wick/Candlestick Autosave Recovery Script
(async function() {
    console.log('🔍 Scanning for autosaves...');
    
    // Try to access Wick's autosave system directly
    if (typeof Wick !== 'undefined' && Wick.AutoSave) {
        console.log('✅ Found Wick.AutoSave API');
        
        Wick.AutoSave.getAutosavesList(function(list) {
            console.log('📁 Found ' + list.length + ' autosave(s):');
            console.table(list);
            
            if (list.length > 0) {
                console.log('\\n📥 To download an autosave, run:');
                console.log('downloadAutosave("' + list[0].uuid + '")');
            }
        });
        
        // Add download helper function
        window.downloadAutosave = function(uuid) {
            Wick.AutoSave.readAutosaveData(uuid, function(data) {
                const blob = new Blob([JSON.stringify(data)], {type: 'application/json'});
                const url = URL.createObjectURL(blob);
                const a = document.createElement('a');
                a.href = url;
                a.download = 'wick-autosave-' + uuid.substring(0,8) + '.json';
                a.click();
                URL.revokeObjectURL(url);
                console.log('✅ Downloaded! Import this JSON file into the recovery tool.');
            });
        };
        
        window.downloadAllAutosaves = function() {
            Wick.AutoSave.getAutosavesList(function(list) {
                list.forEach(function(item, i) {
                    setTimeout(function() {
                        downloadAutosave(item.uuid);
                    }, i * 500);
                });
            });
        };
        
        console.log('\\n💡 Helper functions added:');
        console.log('  downloadAutosave("uuid") - Download single autosave');
        console.log('  downloadAllAutosaves() - Download all autosaves');
        
    } else {
        console.log('❌ Wick.AutoSave not found. Scanning IndexedDB directly...');
        
        // List all databases
        const databases = await indexedDB.databases();
        console.log('📦 Found ' + databases.length + ' database(s):');
        console.table(databases);
    }
})();
